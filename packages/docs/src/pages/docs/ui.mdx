## Subtitle

```js
OUI({
  subtitle: {
    source: [
      {
        name: 'en',
        src: 'your-subtitle-file-path',
        default: localStorage.getItem('pre-subtitle') == 'your-subtitle-file-path'
      },
      {
        name: 'en-us',
        src: 'your-subtitle-file-path',
        default: localStorage.getItem('pre-subtitle') == 'your-subtitle-file-path'
      }
    ],
    offset: 0, // delay any seconds
    onChange(source) {
      // persistedstate
      localStorage.setItem('pre-subtitle', source.src)

      if (source.src == 'abc') {
        // or custom fetch
        return yourFetch(`https://abc.com/subtile/${source.src}`, { mode: 'cors' })
      }
    }
  }
})
```

## Methods

All methods on `player.pluginManager`

```js
// - Update subtitle
const ui = player.pluginManager.getPlugin('ui')
ui.subtitle.changeSource([])

// - Update highlight
ui.changeHighlightSource([])

// - Update thumbnails
ui.changeThumbnails('src')

// - Register menu
player.pluginManager.menus.register({})
player.pluginManager.menus.unregister('name')
player.pluginManager.menus.select('name', 'index')

// - Register setting
player.pluginManager.settings.register({})
player.pluginManager.settings.unregister('key')
player.pluginManager.settings.updateLabel('key', 'text')
player.pluginManager.settings.select('key', 'index')

// - Display error,notice
player.emit('error', { message: 'msg', code: 'number' })
player.emit('notice', { text: 'hello' })
```

## Custom UI

You can build your own UI using `createUI()` from `@oplayer/core`. It provides a simple subscription-based API to receive settings and menus from plugins.

```js
const { Player, createUI } = OPlayer

const myUI = createUI((ctx) => {
  const { player, settings, menus, $video } = ctx

  const $root = document.createElement('div')
  $root.className = 'my-ui'
  $root.innerHTML = `
    <button class="play">▶</button>
    <div class="menus"></div>
    <div class="settings"></div>
  `

  // Play control
  $root.querySelector('.play').onclick = () => player.togglePlay()

  // Subscribe to menus registered by plugins (Chromecast, AirPlay, etc.)
  menus.onRegister((def) => {
    const $btn = document.createElement('button')
    $btn.textContent = def.name
    $btn.onclick = () => def.onClick?.($btn, player)
    $root.querySelector('.menus').appendChild($btn)
  })

  // Subscribe to settings registered by plugins (HLS Quality, Danmaku, etc.)
  settings.onRegister((def) => {
    const $btn = document.createElement('button')
    $btn.textContent = def.name
    $btn.onclick = () => settings.select(def.key, true)
    $root.querySelector('.settings').appendChild($btn)
  })

  // Insert next to video
  $video.parentNode.insertBefore($root, $video.nextSibling)

  return { $root }
})

const player = Player.make('#player', { source: { src: 'video.mp4' } })
player.use([myUI()]).create()
```
